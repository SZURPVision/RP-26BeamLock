include(FetchContent)
FetchContent_Declare(mcap
    GIT_REPOSITORY https://github.com/foxglove/mcap.git
    GIT_TAG releases/cpp/v2.1.3
)
FetchContent_GetProperties(mcap)
if(NOT mcap_POPULATED)
    FetchContent_Populate(mcap)
endif()

INCLUDE(FindProtobuf)
FIND_PACKAGE(Protobuf 3 REQUIRED)
INCLUDE_DIRECTORIES(${PROTOBUF_INCLUDE_DIR} ${Protobuf_INCLUDE_DIRS})
find_package(OpenCV REQUIRED)

set(raw_image_proto ${mcap_SOURCE_DIR}/cpp/examples/protobuf/proto/foxglove/RawImage.proto)
set(proto_generated_dir ${CMAKE_CURRENT_BINARY_DIR}/autogenerated)
set(proto_sources
    ${proto_generated_dir}/RawImage.pb.cc
    ${proto_generated_dir}/RawImage.pb.h
)

add_custom_command(
    OUTPUT ${proto_sources}
    BYPRODUCTS ${proto_sources}
    COMMAND ${CMAKE_COMMAND} -E make_directory autogenerated
    COMMAND ${Protobuf_PROTOC_EXECUTABLE}
            --proto_path=${mcap_SOURCE_DIR}/cpp/examples/protobuf/proto/foxglove
            --cpp_out=${proto_generated_dir}
            ${raw_image_proto}
    DEPENDS ${raw_image_proto}
)

add_library(foxglove_proto_lib ${proto_sources})
target_link_libraries(foxglove_proto_lib PUBLIC ${Protobuf_LIBRARIES})
target_include_directories(foxglove_proto_lib PUBLIC ${proto_generated_dir})

target_compile_options(foxglove_proto_lib PUBLIC -Wno-error=unused-parameter)

add_executable(export Export.cpp)
target_include_directories(export PRIVATE ${mcap_SOURCE_DIR}/cpp/mcap/include)
target_link_libraries(export PRIVATE foxglove_proto_lib ${OpenCV_LIBS})
target_compile_definitions(export PUBLIC MCAP_COMPRESSION_NO_LZ4)
target_compile_definitions(export PUBLIC MCAP_COMPRESSION_NO_ZSTD)
